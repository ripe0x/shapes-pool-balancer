contract StressB is IndependentB {
 function testStressNineConcurrentDrawsKeepersOnlyAndCloseHold() public {start(board(),5 ether,2500);uint[] memory a=buy(buyer,4);uint[] memory b=buy(stranger,5);assertEq(pool.unsettledAcquisitionCount(),9,"hook is checked once per FWA batch");assertEq(ServiceB(VRF_SERVICE).coordinatorOutstandingFor(address(pool)),9);vm.prank(creator);balancer.close();uint retained=balancer.requiredReserve();assertEq(retained,9*0.00146 ether);uint free=balancer.withdrawableSurplus();vm.prank(creator);balancer.claimSurplus(free,creator);for(uint i=5;i>0;--i)fulfill(b[i-1],i);for(uint i=4;i>0;--i)fulfill(a[i-1],i+10);assertEq(ServiceB(VRF_SERVICE).coordinatorOutstandingFor(address(pool)),0);quiesce();assertEq(pool.unsettledAcquisitionCount(),0);assertEq(balancer.reconciledSequence(),9);assertEq(balancer.requiredReserve(),0);emit log_named_uint("nineDrawClosedHold",retained);unwind();}
 function testStressStoredServiceWordRedeliveryAndReplay() public {start(20,1 ether,2500);uint w=wordForSlot(2);uint[] memory req=buy(buyer,1);uint[] memory words=new uint[](1);words[0]=w;address coordinator=IVRFService(VRF_SERVICE).coordinator();vm.roll(block.number+3);vm.prank(coordinator);IVRFService(VRF_SERVICE).rawFulfillRandomWords{gas:500000}(req[0],words);assertEq(ServiceB(VRF_SERVICE).requestStatus(req[0]),2,"word stored despite no delivery gas");assertEq(ServiceB(VRF_SERVICE).requestPool(req[0]),address(pool));assertEq(ServiceB(VRF_SERVICE).storedRandomWord(req[0]),w);assertEq(ServiceB(VRF_SERVICE).coordinatorOutstandingFor(address(pool)),0);vm.prank(stranger);assertTrue(ServiceB(VRF_SERVICE).deliverRandomness(req[0]));quiesce();assertEq(pool.unsettledAcquisitionCount(),0);uint balance=address(balancer).balance;words[0]=0;vm.prank(coordinator);IVRFService(VRF_SERVICE).rawFulfillRandomWords{gas:500000}(req[0],words);assertEq(ServiceB(VRF_SERVICE).requestStatus(req[0]),4);assertEq(address(balancer).balance,balance);unwind();}
 function testStressLowGasPaidAttemptCannotGetPaidForFailure() public {start(board(),5 ether,2500);uint[] memory req=buy(buyer,2);fulfill(req[1],202);fulfill(req[0],101);uint keeperBefore=keeper.balance;uint sequence=balancer.reconciledSequence();bytes32 h=balancer.settingsHash();vm.prank(keeper);(bool ok,)=address(balancer).call{gas:100000}(abi.encodeWithSignature("performUpkeep(bytes32,uint256)",h,0));assertFalse(ok);assertEq(keeper.balance,keeperBefore);assertEq(balancer.reconciledSequence(),sequence);quiesce();assertEq(pool.unsettledAcquisitionCount(),0);unwind();}
 function testEconomicHostilePolicyCreatorBuyerKeeperCoalition() public {fees(0.01 gwei);start(20,2 ether,2500);HostilePolicyB p=new HostilePolicyB();vm.prank(creator);balancer.proposeKeeperPolicy(address(p));vm.warp(block.timestamp+2 days);balancer.activateKeeperPolicy();uint before=creator.balance+buyer.balance+keeper.balance+address(balancer).balance+0.2 ether;uint edgeBefore=MoneyB(address(pool)).accruedProtocolFees();(,uint vrf,)=pool.quoteAcquisitionPrice(1);uint w=wordForSlot(2);uint[] memory req=buy(buyer,1);fulfill(req[0],w);quiesce();(,,,uint lid,uint8 status,)=pool.acquisitions(req[0]);assertEq(status,2);(,,,uint token,,,,)=pool.listings(lid);assertEq(shapes.backingOf(token),0.01 ether);assertEq(balancer.getActiveShapes()[0],20);uint afterAssets=creator.balance+buyer.balance+keeper.balance+address(balancer).balance+0.21 ether;uint edge=MoneyB(address(pool)).accruedProtocolFees()-edgeBefore;assertEq(before-afterAssets,edge+vrf+0.001 ether);emit log_named_uint("coalitionLostExcludingTransactionGas",before-afterAssets);emit log_named_uint("FWAedge",edge);emit log_named_uint("VRFfee",vrf);emit log_named_uint("mintFee",0.001 ether);unwind();}
 function testFuzzReachableAccountingAfterConcurrentActions(uint64 seed) public {start(30,2 ether,uint16(1000+uint(seed)%9001));uint[] memory req=buy(buyer,1+uint(seed)%4);uint[] memory more=buy(stranger,1+uint(seed>>8)%5);assertGe(pool.unsettledAcquisitionCount(),2);for(uint i=req.length;i>0;--i)fulfill(req[i-1],uint(keccak256(abi.encode(seed,i))));for(uint i=more.length;i>0;--i)fulfill(more[i-1],uint(keccak256(abi.encode(seed,i+77))));quiesce();assertEq(pool.unsettledAcquisitionCount(),0);assertLe(balancer.profit(),address(balancer).balance);assertEq(pool.ownerProceeds(),0);assertEq(MoneyB(address(pool)).acquisitionEscrowTotal(),0);assertGe(address(pool).balance,MoneyB(address(pool)).deliveryEscrowTotal()+MoneyB(address(pool)).acquisitionRefundCreditTotal()+MoneyB(address(pool)).accruedProtocolFees());vm.prank(creator);balancer.close();assertEq(balancer.requiredReserve(),0);unwind();}
